# Caddyfile for custom subdomain + automatic Let's Encrypt
git.example.com {
  handle /api/* { reverse_proxy 127.0.0.1:3001 }
  handle /git/* { reverse_proxy 127.0.0.1:3001 }
  handle {
    root * /opt/nixre/ui/dist
    try_files {path} /index.html
    file_server
  }
}
